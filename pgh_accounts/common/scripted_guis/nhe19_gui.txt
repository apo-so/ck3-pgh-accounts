nhe19_ui_visible = {
 scope = character
 is_shown = { has_variable = nhe19_ui_open }
}

nhe19_ui_valid_view = {
 scope = character
 is_shown = { exists = var:nhe19_view_title }
}

nhe19_ui_close = {
 scope = character
 is_shown = { always = yes }
 is_valid = {  }
 effect = { remove_variable = nhe19_ui_open }
}

nhe19_ui_next = {
 scope = character
 is_shown = { always = yes }
 is_valid = {  }
 effect = { change_variable = { name = nhe19_view_index add = 1 } nhe19_refresh_view_effect = yes }
}

nhe19_ui_previous = {
 scope = character
 is_shown = { always = yes }
 is_valid = {  }
 effect = { change_variable = { name = nhe19_view_index add = -1 } nhe19_refresh_view_effect = yes }
}

}
