 holder = scope:nhe45_actor
 NOT = { this = scope:nhe45_keep }
 NOT = { this = scope:nhe45_primary }
 exists = scope:nhe45_keep
 scope:nhe45_keep = { nhe45_available_carrier_trigger = yes holder = scope:nhe45_actor }
 exists = scope:nhe45_primary
 scope:nhe45_primary = { holder = scope:nhe45_actor }
 exists = scope:nhe45_default
 scope:nhe45_default = { owner = scope:nhe45_actor }
 scope:nhe45_actor = { primary_title = scope:nhe45_primary domicile = scope:nhe45_default }
 NOT = { title_domicile = scope:nhe45_default }

 # A canonical object promised to a different office holder must first finish its handoff.

  var:nhe19_source_office = { holder = scope:nhe45_actor }
 }
}
